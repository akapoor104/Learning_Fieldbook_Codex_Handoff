To repeat the home procedure, set the first line of the script after `set -euo pipefail` to the path to a clean checkout of the candidate. The procedure uses disposable data and the standard home Compose definition. Local ports 18086 and a Docker-assigned database port are used.

```bash
#!/usr/bin/env bash
set -euo pipefail
candidate=$(cat /tmp/fieldbook-t0-candidate-path)
cd "$candidate"
printf 'candidate %s\n' "$(git rev-parse HEAD)"
test -z "$(git status --porcelain)"
scratch=$(mktemp -d /tmp/fieldbook-t0-home.XXXXXX)
project=fieldbook-t0-${scratch##*.}
project=${project,,}
sql_container="$project-rust-db"
export FIELD_IMAGE="fieldbook:$project"
export FIELDBOOK_HTTP_PORT=18086
password=$(openssl rand -hex 24)
printf 'FIELDBOOK_DB_PASSWORD=%s\n' "$password" > "$scratch/compose.env"
printf 'POSTGRES_PASSWORD=%s\nPOSTGRES_USER=fieldbook\nPOSTGRES_DB=fieldbook_integration\n' "$password" > "$scratch/sql.env"
compose() { docker compose --env-file "$scratch/compose.env" --project-name "$project" "$@"; }
cleanup() {
    compose down --volumes
    if docker container inspect "$sql_container" >/dev/null 2>&1; then docker stop "$sql_container" >/dev/null; fi
    rm -f "$scratch/compose.env" "$scratch/sql.env"
}
trap cleanup EXIT
psql_fixture() {
    compose exec -T db sh -c 'PGPASSWORD="$POSTGRES_PASSWORD" psql -XAtq -v ON_ERROR_STOP=1 -h 127.0.0.1 -U fieldbook -d fieldbook "$@"' sh "$@"
}
probe() { python3 tools/probe_health.py http://127.0.0.1:18086 "$@"; }

docker compose --env-file .env.example config --quiet
for volume in postgres-data media-data; do
    if docker volume ls --format '{{.Name}}' | grep -Fx "${project}_${volume}"; then exit 1; fi
    printf 'Empty install confirmed: no %s volume\n' "$volume"
done
compose up -d --wait db
compose build web
compose up -d --no-build web
curl --retry 5 --retry-connrefused --retry-delay 1 -fsS http://127.0.0.1:18086/healthz >/dev/null
probe --expect-not-ready
code=$(curl --silent --show-error --output "$scratch/unknown.body" --write-out '%{http_code}' http://127.0.0.1:18086/does-not-exist)
test "$code" = 404
printf 'Unknown route HTTP %s\n' "$code"
compose run --rm --pull never web fieldbook migrate
compose run --rm --pull never web fieldbook migrate
compose up -d --wait --no-build web
probe
test "$(psql_fixture -c 'SELECT version, success FROM _sqlx_migrations;')" = '1|t'
psql_fixture -c 'SHOW server_version;'
docker image inspect "$FIELD_IMAGE" --format '{{.Id}} {{.Os}}/{{.Architecture}} {{.Config.User}}'
fixture_label=$(python3 -c 'import json; print("Fixture household " + json.load(open("fixtures/household.json"))["households"][0])')
test "$fixture_label" = 'Fixture household A'
fixture_id=$(psql_fixture -v "label=$fixture_label" <<'SQL'
INSERT INTO households (label) VALUES (:'label') RETURNING id;
SQL
)
printf 'Saved fixture %s|%s\n' "$fixture_id" "$fixture_label"
printf '%s\n' 'T0 fictional media marker' > "$scratch/original.txt"
compose exec -T web sh -c 'cat > /var/data/fieldbook/t0.txt' < "$scratch/original.txt"
compose stop db
probe --expect-not-ready
compose start db
compose up -d --wait --no-build web
probe
if compose logs --no-color | grep -Fq "$password"; then exit 1; fi
printf 'No disposable password in service logs\n'
compose down
compose up -d --wait --no-build web
probe
restored=$(psql_fixture -v "fixture_id=$fixture_id" <<'SQL'
SELECT id || '|' || label FROM households WHERE id = :'fixture_id';
SQL
)
test "$restored" = "$fixture_id|$fixture_label"
printf 'Restored fixture %s\n' "$restored"
compose exec -T web cat /var/data/fieldbook/t0.txt > "$scratch/restored.txt"
cmp "$scratch/original.txt" "$scratch/restored.txt"
sha256sum "$scratch/original.txt" "$scratch/restored.txt"
for mode in tmp image tmpfs; do
    options=(-e FIELD_STORAGE=filesystem -e FIELD_MEDIA_ROOT=/var/data/fieldbook)
    case "$mode" in
        tmp) options=(-e FIELD_STORAGE=filesystem -e FIELD_MEDIA_ROOT=/tmp/fieldbook) ;;
        tmpfs) options+=(--tmpfs /var/data/fieldbook) ;;
    esac
    status=0
    output=$(docker run --rm "${options[@]}" "$FIELD_IMAGE" fieldbook serve 2>&1) || status=$?
    test "$status" -eq 1
    [[ "$output" == *'requires a read-write persistent mount'* ]]
    printf 'Media mode %s rejected with exit %s\n' "$mode" "$status"
done

docker run --rm --detach --name "$sql_container" --env-file "$scratch/sql.env" --publish 127.0.0.1::5432 postgres:16-bookworm >/dev/null
for attempt in {1..25}; do
    if docker exec "$sql_container" pg_isready -h 127.0.0.1 -U fieldbook -d fieldbook_integration; then break; fi
    sleep 1
done
db_address=$(docker port "$sql_container" 5432/tcp)
FIELDBOOK_TEST_DATABASE_URL="postgres://fieldbook:$password@$db_address/fieldbook_integration" cargo test --workspace --locked -- --include-ignored
```
